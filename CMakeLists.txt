cmake_minimum_required(VERSION 3.16)
project(FlashCoreSSD VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FLASHCORE_SANITIZE "Enable supported address/undefined sanitizers" OFF)
option(FLASHCORE_BUG_TRIM "Deliberate stale-valid-page TRIM bug for RCA" OFF)
if(MSVC)
 add_compile_options(/W4 /permissive-)
 if(FLASHCORE_SANITIZE)
  add_compile_options(/fsanitize=address)
 endif()
else()
 add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
 if(FLASHCORE_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
 endif()
endif()
add_library(flashcore_lib src/flashcore.cpp)
target_include_directories(flashcore_lib PUBLIC include)
if(FLASHCORE_BUG_TRIM)
 target_compile_definitions(flashcore_lib PRIVATE FLASHCORE_BUG_TRIM)
endif()
add_executable(flashcore src/main.cpp)
target_link_libraries(flashcore PRIVATE flashcore_lib)
add_executable(flashcore_tests tests/tests.cpp)
target_link_libraries(flashcore_tests PRIVATE flashcore_lib)
enable_testing()
set(CASES lifecycle erase_guard geometry mapping overwrite trim gc_empty gc_mixed gc_preserve wear invalid_address out_of_space bad_block program_failure gc_program_failure gc_erase_failure retirement retirement_failure counters rebuild random_greedy random_adaptive tie_break pressure adaptive_choice decay partial_failure_recovery rebuild_highest_sequence active_exclusion retire_active retire_no_reserve nand_fault_atomicity trim_metrics adaptive_divergence)
foreach(case IN LISTS CASES)
 add_test(NAME ${case} COMMAND flashcore_tests ${case})
endforeach()
add_test(NAME cli_demo COMMAND flashcore demo)
add_test(NAME cli_invalid COMMAND flashcore benchmark --policy wrong)
set_tests_properties(cli_invalid PROPERTIES WILL_FAIL TRUE)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
 add_test(NAME python_contracts COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/python/test_validation.py $<TARGET_FILE:flashcore>)
endif()
add_executable(fault_campaign tests/fault_campaign.cpp)
target_link_libraries(fault_campaign PRIVATE flashcore_lib)
add_test(NAME fault_campaign COMMAND fault_campaign)
